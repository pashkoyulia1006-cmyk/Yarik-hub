    <div id="create-group">
        <h3>Створити свою Спільноту</h3>
        <input type="text" id="groupName" placeholder="Назва групи (напр. Гільдія Магів)">
        <textarea id="groupDesc" placeholder="Опис групи"></textarea>
        <button onclick="createGroup()">Створити</button>
    </div>

    <hr>

    <div id="groupsList"></div>

    <script>
        // Завантаження груп з сервера
        async function loadGroups() {
            const response = await fetch('/api/groups');
            const groups = await response.json();
            const list = document.getElementById('groupsList');
            list.innerHTML = '';
            groups.forEach(g => {
                list.innerHTML += 
                    <div class="group-card">
                        <h3>${g.name}</h3>
                        <p>${g.description}</p>
                    </div>
                ;
            });
        }

        // Надсилання нової групи
        async function createGroup() {
            const name = document.getElementById('groupName').value;
            const description = document.getElementById('groupDesc').value;

            await fetch('/api/groups', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ name, description })
            });

            loadGroups(); // Оновити список
        }

        loadGroups(); // Викликати при старті
    </script>
